cmake_minimum_required(VERSION 3.16)
project(XrpcInstalledConsumer LANGUAGES CXX)
find_package(Threads REQUIRED)
# What an installation provides differs by distribution: the udp component alone
# on Bionic, the others on focal and newer. Select the components to check.
option(XRPC_CHECK_HTTP "Check the installed json_http component (consumer)" ON)
option(XRPC_CHECK_GRPC "Also require the installed optional gRPC component (needs XRPC_CHECK_HTTP)" OFF)
option(XRPC_CHECK_UDP "Check the installed udp component (udp_consumer)" OFF)
if(XRPC_CHECK_GRPC AND NOT XRPC_CHECK_HTTP)
  message(FATAL_ERROR "The consumer links gRPC together with json_http")
endif()
set(_xrpc_probe_components "")
if(XRPC_CHECK_HTTP)
  list(APPEND _xrpc_probe_components json_http)
endif()
if(XRPC_CHECK_GRPC)
  list(APPEND _xrpc_probe_components grpc)
endif()
if(XRPC_CHECK_UDP)
  list(APPEND _xrpc_probe_components udp)
endif()
if(NOT _xrpc_probe_components)
  message(FATAL_ERROR "Select at least one component to check")
endif()
find_package(XgcXrpc 0.1.0 CONFIG REQUIRED COMPONENTS ${_xrpc_probe_components})
if(XRPC_CHECK_HTTP)
  add_executable(consumer consumer.cpp)
  target_compile_features(consumer PRIVATE cxx_std_17)
  target_link_libraries(consumer PRIVATE XgcXrpc::json_http Threads::Threads)
  if(XRPC_CHECK_GRPC)
    target_compile_definitions(consumer PRIVATE XRPC_CHECK_GRPC=1)
    target_link_libraries(consumer PRIVATE XgcXrpc::grpc)
  endif()
endif()
if(XRPC_CHECK_UDP)
  add_executable(udp_consumer udp_consumer.cpp)
  target_compile_features(udp_consumer PRIVATE cxx_std_17)
  target_link_libraries(udp_consumer PRIVATE XgcXrpc::udp Threads::Threads)
endif()
