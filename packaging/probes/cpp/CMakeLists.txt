cmake_minimum_required(VERSION 3.16)
project(XrpcInstalledConsumer LANGUAGES CXX)
find_package(Threads REQUIRED)
option(XRPC_CHECK_GRPC "Require the installed optional gRPC component" OFF)
if(XRPC_CHECK_GRPC)
  find_package(XgcXrpc 0.1.0 CONFIG REQUIRED COMPONENTS json_http grpc)
else()
  find_package(XgcXrpc 0.1.0 CONFIG REQUIRED COMPONENTS json_http)
endif()
add_executable(consumer consumer.cpp)
target_compile_features(consumer PRIVATE cxx_std_20)
target_link_libraries(consumer PRIVATE XgcXrpc::json_http Threads::Threads)
if(XRPC_CHECK_GRPC)
  target_compile_definitions(consumer PRIVATE XRPC_CHECK_GRPC=1)
  target_link_libraries(consumer PRIVATE XgcXrpc::grpc)
endif()
